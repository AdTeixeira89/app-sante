rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    function isSignedIn() {
      return request.auth != null;
    }

    function isMember(familyId) {
      return isSignedIn() &&
        request.auth.uid in get(/databases/$(database)/documents/families/$(familyId)).data.members;
    }

    // Documento da própria família
    match /families/{familyId} {
      allow get: if isMember(familyId);
      allow list: if false;

      allow create: if isSignedIn()
        && request.resource.data.members.size() == 1
        && request.resource.data.members[0] == request.auth.uid
        && request.resource.data.ownerUid == request.auth.uid;

      allow update: if isMember(familyId);
      allow delete: if false;

      // Pedidos de acesso: qualquer pessoa autenticada pode pedir para si própria;
      // só um membro já aprovado pode ler ou apagar pedidos (aprovar/recusar)
      match /joinRequests/{uid} {
        allow create: if isSignedIn() && request.auth.uid == uid;
        allow read, delete: if isMember(familyId);
      }

      // Tudo o resto dentro da família (rdvs, meds, docs, medLog, meta, presence...)
      match /{document=**} {
        allow read, write: if isMember(familyId);
      }
    }

    // Códigos de convite: pesquisáveis por qualquer pessoa autenticada,
    // mas nunca listáveis (impede varrer todos os códigos existentes)
    match /inviteCodes/{code} {
      allow get: if isSignedIn();
      allow create: if isSignedIn();
      allow list: if false;
    }

    // Índice utilizador -> família (para saber a que família cada conta pertence)
    match /userFamilies/{uid} {
      allow read: if isSignedIn() && request.auth.uid == uid;
      allow write: if isSignedIn() &&
        (request.auth.uid == uid || isMember(request.resource.data.familyId));
    }
  }
}
